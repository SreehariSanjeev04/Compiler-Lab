decl
  int m, n;
  int ack(int m, int n);
enddecl

int ack(int m, int n) {
  decl
    int r;
  enddecl
  begin
    if(m == 0) then
      r = n + 1;
    else
      if(n == 0) then
        r = ack(m - 1, 1);
      else
        r = ack(m - 1, ack(m, n - 1));
      endif;
    endif;
    return r;
  end
}

int main() {
  decl
    int r;
  enddecl
  begin
    read(m);
    read(n);
    r = ack(m, n);
    write(r);
    return 0;
  end
}
