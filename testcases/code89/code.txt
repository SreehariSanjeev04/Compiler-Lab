class
  A {
    decl
      int x;
      int mix(int a, int b);
      int both(int a, int b);
    enddecl

    int mix(int a, int b) {
      begin
        return self.x + a * 10 + b;
      end
    }

    int both(int a, int b) {
      begin
        return self.mix(a, b) + self.mix(b, a);
      end
    }
  }
endclass

decl
  A p;
enddecl

int main() {
  begin
    initialize();
    p = new(A);
    p.x = 4;
    write(p.mix(2, 3));
    write(p.mix(p.mix(1, 2), 3));
    write(p.both(2, 3));
    return 0;
  end
}
