decl
    int m[3][3];
enddecl

int main() {
  decl
    int i, j;
  enddecl

  begin
i = 0;
while (i < 3) do
    j = 0;
    while (j < 3) do
        m[i][j] = (i + 1) * (j + 1);
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;

i = 0;
while (i < 3) do
    j = 0;
    while (j < 3) do
        write(m[i][j]);
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;

write(m[0][0] + m[1][1] + m[2][2]);
  end
}
