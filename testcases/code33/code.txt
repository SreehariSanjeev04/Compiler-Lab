begin
decl
    int a[2][2];
    int b[2][2];
    int c[2][2];
    int s, i, j, k;
enddecl
i = 0;
while (i < 2) do
    j = 0;
    while (j < 2) do
        read(a[i][j]);
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;

i = 0;
while (i < 2) do
    j = 0;
    while (j < 2) do
        read(b[i][j]);
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;

i = 0;
while (i < 2) do
    j = 0;
    while (j < 2) do
        s = 0;
        k = 0;
        while (k < 2) do
            s = s + a[i][k] * b[k][j];
            k = k + 1;
        endwhile;
        c[i][j] = s;
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;

i = 0;
while (i < 2) do
    j = 0;
    while (j < 2) do
        write(c[i][j]);
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;
end;
