decl
    str s[2][3];
    str *a;
enddecl

int main() {
  decl
    int i, j;
  enddecl

  begin
s[0][0] = "AA";
s[0][1] = "BB";
s[0][2] = "CC";
s[1][0] = "DD";
s[1][1] = "EE";
s[1][2] = "FF";
write(**s);
write(*s[1]);
write(*(s[1] + 1));
a = s[0];
write(*a);
a = &s[1][2];
write(*a);
*(a - 1) = "NN";
write(*a);
write(s[1][1]);
a = s[0];
*(a + 2) = "ZZ";
write(*(a + 2));
write(s[0][2]);
i = 0;
while (i < 2) do
    j = 0;
    while (j < 3) do
        write(s[i][j]);
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;
  end
}
