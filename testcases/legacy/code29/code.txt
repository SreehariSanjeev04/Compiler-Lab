begin
decl
    int a[3][3];
    int rs, cs, i, j;
enddecl
i = 0;
while (i < 3) do
    j = 0;
    while (j < 3) do
        read(a[i][j]);
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;

i = 0;
while (i < 3) do
    rs = 0;
    j = 0;
    while (j < 3) do
        rs = rs + a[i][j];
        j = j + 1;
    endwhile;
    write(rs);
    i = i + 1;
endwhile;

j = 0;
while (j < 3) do
    cs = 0;
    i = 0;
    while (i < 3) do
        cs = cs + a[i][j];
        i = i + 1;
    endwhile;
    write(cs);
    j = j + 1;
endwhile;
end;
