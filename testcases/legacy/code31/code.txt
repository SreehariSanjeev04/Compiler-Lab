begin
decl
    int a[2][3];
    int b[3][2];
    int i, j;
enddecl
i = 0;
while (i < 2) do
    j = 0;
    while (j < 3) do
        read(a[i][j]);
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;

i = 0;
while (i < 2) do
    j = 0;
    while (j < 3) do
        b[j][i] = a[i][j];
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;

i = 0;
while (i < 3) do
    j = 0;
    while (j < 2) do
        write(b[i][j]);
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;
end;
