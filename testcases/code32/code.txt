decl
    int a[2][2];
enddecl

int main() {
  decl
    int t, s, i, j;
  enddecl

  begin
i = 0;
while (i < 2) do
    j = 0;
    while (j < 2) do
        read(a[i][j]);
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;

s = 0;
i = 0;
while (i < 2) do
    j = 0;
    while (j < 2) do
        s = s + a[i][j];
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;
write(s);

t = a[0][1];
a[0][1] = a[1][0];
a[1][0] = t;
write(a[0][1]);
write(a[1][0]);
write(a[0][0]);
write(a[1][1]);
  end
}
