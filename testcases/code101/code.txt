class
A {
decl
int f(int x);
enddecl
int f(int x) {
begin
    return x+1;
end
}
}
B extends A {
decl
int f(int x, int y);
enddecl
int f(int x, int y) {
begin
    return x*10+y;
end
}
}
endclass
decl
B b;
enddecl
int main() {
decl
    int status;
enddecl
begin
status=initialize();
b=new(B);
write(b.f(5));
write(b.f(2,3));
return 0;
end
}
