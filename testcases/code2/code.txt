decl
  int n, factorial(int m);
enddecl

int factorial(int m) {
  decl
    int f;
  enddecl
  begin
    if(m <= 1) then
      f = 1;
    else
      f = m * factorial(m-1);
    endif;
    return f;
  end
}

    int r;
  enddecl
