class
  A {
    decl
      int f0;
      int f1;
      int f2;
      int f3;
      int f4;
      int f5;
      int f6;
      int f7;
    enddecl
  }
endclass

decl
  A p;
enddecl

    initialize();

    p.f7 = 8;
    write(p.f7);

    p = new(A);
    write(p.f7);

  end
